#include<iostream>
using namespace std;
#include<stdlib.h>
class Node
{
    public:
	int data;
	Node *next;
};
class CirList
{
	Node *tail;
	public:
	CirList()
	{
		tail=NULL;
	}
	void createList();
	void insert_front();
	void insert_back();
	void insert_after();
	void delete_front();
	void delete_back();
	void delete_node();
	void traverse();
	void search();
	void update();
	~CirList()
	{
	    Node *t1,*t2;
	    t1=t2=tail->next;
		while(t1!=tail)
        {
            t2=t1;
            t1=t1->next;
            tail->next=t1;
            delete t2;
        }
        delete t1;
        tail=NULL;
	}
};
void CirList::createList()
{
    char choice;
   Node *newNode,*nn;

      cin>>choice;
       if(choice=='n')
        break;
       else
       {
            nn=new Node;
            cout<<"Enter data:";
            cin>>nn->data;
            nn->next=tail->next;
            newNode->next=nn;
            tail=nn;
       }
       newNode=nn;
   }
}
void CirList::insert_front()
{
	Node *nn=new Node();
	cout<<"\nEnter data:";
	cin>>nn->data;
	if(tail==NULL)
	{
		tail=nn;
		nn->next=tail;
	}
	else
	{
		nn->next=tail->next;
		tail->next=nn;
	}
}
void CirList::insert_back()
{

	cout<<"\nEnter data:";

	{
		tail=nn;
		nn->next=tail;
	}
	else
	{
		nn->next=tail->next;

}
void CirList::insert_after()
{
	int nnum,i;
	Node *nn=new Node();
	cout<<"\nEnter data:";
	cin>>nn->data;
	cout<<"\nEnter node Number:";
	cin>>nnum;
	Node *t;
	t=tail->next;
	for(i=1;i<nnum;i++)
	{
		t=t->next;
		if(t==tail->next)
		{
			cout<<nnum<<" node does not exist\n";
			break;
		}
	}
	if(t==tail)
	{
		nn->next=tail->next;
		tail->next=nn;
		tail=nn;
	}
	else
	{
		nn->next=t->next;
		t->next=nn;
	}
}
void CirList::delete_front()
{
	Node *t;
	t=tail->next;
	if(tail->next==tail)
	{
		free(t);
		tail=NULL;
	}
	else
	{
		tail->next=t->next;
		free(t);
	}
}
void CirList::delete_back()
{
	Node *t1,*t2;
	t1=t2=tail->next;
	if(t1==tail)
	{
		free(tail);
		tail=NULL;
	}
	else
	{
		while(t1->next!=tail)
		{
			t1=t1->next;
		}
		free(tail);
		tail=t1;
		tail->next=t2;
	}
}

void CirList::delete_node()
{

	cout<<"\nEnter node number which you want to delete?";
	cin>>nnum;
	Node *t1,*t2;
	t1=t2=tail->next;
	if(nnum==1)//IF FIRST NODE IS SELECTED
	{
	    if(tail==tail->next)
        {
            free(t1);
            free(t2);
            tail=NULL;
        }
        else
        {
                t1=t1->next;
                tail->next=t1;
                free(t2);
        }

	}
	else
	{

		for(i=1;i<nnum;i++)//IF OTHER THAN FIRST & LAST NODES ARE SELECTED
		{
			t2=t1;

		}
		t2->next=t1->next;
		if(t1==tail)//IF LAST NODE IS SELECTED
        	{
            		tail=t2;
        	}
       		free(t1);
	}
}
void CirList::traverse()
{
	Node *t,*h;
	if(tail==NULL)
    	{
        cout<<"\nList empty";
    	}
    	else
    	{
        	t=h=tail->next;
        	do
		{
			cout<<t->data<<" ";
			t=t->next;

		}while(t!=h);
    	}
}
void CirList::search()
{
    int data,pos=0,fl=0;
    cout<<"\nEnter the data to be searched:";
    cin>>data;
	Node *t,*h;
	if(tail==NULL)
    	{
	        cout<<"\nList empty";
	}
	else
	{
	        t=h=tail->next;
	        do
		{
		    pos=pos+1;
		    if(data==t->data)
	            {
	                fl=1;
	                break;
            	    }
            		t=t->next;
		}while(t!=h);
		
		if(fl==1)
            		cout<<"Search data found in position "<<pos;
        	else
            		cout<<"Search data not found";
    	}
}
void CirList::update()
{
    int data,pos,cp=0,fl=0;
    cout<<"\nEnter the position of the data to be updated:";

    cout<<"\nEnter new data:";
    cin>>data;
	Node *t,*h;
	if(tail==NULL)
    	{
        	cout<<"\nList empty";
    	}
    	else
    	{
       	 t=h=tail->next;
       	 	do
	  	{
			cp=cp+1;
			if(cp==pos)
	            	{
	               	 t->data=data;
	                	fl=1;
	                	break;
	            	}
        	    t=t->next;

	   	}while(t!=h);

		if(fl==1)
            	cout<<"Data Updated";
        	else
           	 cout<<"Given Position is incorrect";
    	}
}
