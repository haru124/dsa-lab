#include<iostream>
using namespace std;
void TowerOfHanoi(int n,char src,char dstn,char aux)
{

    if(n==0)
        return;
    TowerOfHanoi(n-1,src,aux,dstn);
    cout<<src<<":"<<dstn<<endl;
    TowerOfHanoi(n-1,aux,dstn,src);
}

int main()
{
int n;

cin>>n;
TowerOfHanoi(n,'a','c','b');
}
