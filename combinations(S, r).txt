def generate_combinations(S, r):
    results = []

    def backtrack(start, path):
        if len(path) == r:
            results.append(path[:])
            return
        for i in range(start, len(S)):
            path.append(S[i])
            backtrack(i+1, path)  # move forward
            path.pop()

    backtrack(0, [])
    return results

# Example
print("Combinations:", generate_combinations([1,2,3], 2))
# [[1,2],[1,3],[2,3]]
