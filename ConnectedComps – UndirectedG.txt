Connected Components – Undirected Graph

Idea:

Do a DFS from each unvisited node → all nodes reached are in the same connected component.

#include <bits/stdc++.h>
using namespace std;

void dfsUndirected(int node, vector<vector<int>>& adj, vector<bool>& visited, vector<int>& component) {
    visited[node] = true;
    component.push_back(node);
    for (int neigh : adj[node]) {
        if (!visited[neigh]) {
            dfsUndirected(neigh, adj, visited, component);
        }
    }
}

vector<vector<int>> connectedComponentsUndirected(int n, vector<vector<int>>& adj) {
    vector<bool> visited(n, false);
    vector<vector<int>> components;

    for (int i = 0; i < n; i++) {
        if (!visited[i]) {
            vector<int> component;
            dfsUndirected(i, adj, visited, component);
            components.push_back(component);
        }
    }
    return components;
}

int main() {
    int n = 6;
    vector<vector<int>> adj(n);
    adj[0] = {1};
    adj[1] = {0,2};
    adj[2] = {1};
    adj[3] = {4};
    adj[4] = {3};
    adj[5] = {}; // isolated node

    auto comps = connectedComponentsUndirected(n, adj);
    for (auto &c : comps) {
        for (int node : c) cout << node << " ";
        cout << endl;
    }
}
