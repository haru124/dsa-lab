#include<iostream>
using namespace std;
#include<stdlib.h>
typedef struct Node
{
	int data;
	Node *prev;
	Node *next;
};
class DList
{
	Node *head;
	public:
	DList()
	{
		head=NULL;
	}
	void createList();
	void insert_front();
	void insert_back();
	void insert_after();
	void delete_front();
	void delete_back();
	void delete_node();
	void traverse();
	void search();
	void update();
	void reverse();
	~DList()
	{
	    Node *temp;
		while(head!=NULL)
        {
            temp=head;
            head=head->next;
            head->prev=NULL;
            delete temp;
        }
	}
};

        break;

       newNode=nn;
   }
}
void DList::insert_front()
{
	Node *nn=new Node();
	cout<<"\nEnter data:";
	cin>>nn->data;
	nn->prev=NULL;
	if(head==NULL)
	{
		head=nn;
		nn->next=NULL;
	}
	else
	{
		nn->next=head;
		head->prev=nn;
		head=nn;
	}
}
void DList::insert_back()
{
	Node *nn=new Node;
	cout<<"\nEnter data:";
	cin>>nn->data;
	nn->next=NULL;
	if(head==NULL)
	{
		nn->prev=NULL;
		head=nn;
	}
	else
	{
		Node *t;
		t=head;
		while(t->next!=NULL)
		{
			t=t->next;
		}
		nn->prev=t;
		t->next=nn;

	}
}

void DList::delete_front()
{
	Node *t;
	t=head;
	if(t->next==NULL)
	{
		head=NULL;
		free(t);
	}
	else
	{
		head=t->next;
		head->prev=NULL;
		free(t);
	}
}
void DList::delete_back()
{
	Node *t1,*t2;
	t1=t2=head;
	while(t1->next!=NULL)
	{
		t2=t1;
		t1=t1->next;
	}
	if(t1==head)//IF ONLY ONE NODE IS PRESENT
	{
		head=NULL;
		free(t1);
	}
	else
	{
		t2->next=NULL;
		free(t1);
	}
}

}
void DList::traverse()
{
	Node *t;
	t=head;
	if(t==NULL)
		cout<<"\nList empty";
	else
	{       while(t!=NULL)
            {
                cout<<t->data<<" ";
                t=t->next;
            }
	}
}
void DList::search()
{
    int data,fl=0,pos=0;
    cout<<"Enter data need to be searched";
    cin>>data;
    Node *t;
    t=head;
    if(t==NULL)
    {
        cout<<"List is empty";
    }
    else
        {
            while(t!=NULL)
            {
                pos=pos+1;
                if(t->data==data)
                {
                    fl=1;
                    break;
                }
                else
                {
                    t=t->next;
                }
            }
            if(fl==1)
            cout<<"Search data found in the position"<<pos;
            else
            cout<<"Search data not found";
        }
}
void DList::update()
{
    int pos,cp=0,fl=0;
    cout<<"Enter the position of the node to be updated";
    cin>>pos;
    int data;
    cout<<"Enter new data";
    cin>>data;
    Node *t=new Node();
    t=head;
    cp=1;
    if(pos==1)
    {
        t->data=data;
    }
    else
        {
            while(t->next!=NULL)
            {
                t=t->next;
                cp=cp+1;
                if(cp==pos)
                {
                    t->data=data;
                    cout<<"Data updated";
                    fl=1;
                    break;
                }
            }
            if(fl==0)
                cout<<"You have entered position greater then the available position";
        }
}
void DList::reverse()
{
    Node *t=NULL,*currNode;
    currNode=head;
    if(head==NULL)
        cout<<"List empty";
    else
        {
            while(currNode!=NULL)
            {
                t=currNode->prev;
                currNode->prev=currNode->next;
                currNode->next=t;
                currNode=currNode->prev;
            }
            head=t->prev;
        }

}
