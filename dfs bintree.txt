#include <iostream>
using namespace std;

struct TreeNode {
    int val;
    TreeNode* left;
    TreeNode* right;
};

// Inorder (DFS) - Left, Root, Right
void inorder(TreeNode* root) {
    if (!root) return;
    inorder(root->left);
    cout << root->val << " ";
    inorder(root->right);
}

// Preorder (DFS) - Root, Left, Right
void preorder(TreeNode* root) {
    if (!root) return;
    cout << root->val << " ";
    preorder(root->left);
    preorder(root->right);
}

// Postorder (DFS) - Left, Right, Root
void postorder(TreeNode* root) {
    if (!root) return;
    postorder(root->left);
    postorder(root->right);
    cout << root->val << " ";
}

int main() {
    TreeNode* root = new TreeNode{4, nullptr, nullptr};
    root->left = new TreeNode{2, new TreeNode{1,nullptr,nullptr}, new TreeNode{3,nullptr,nullptr}};
    root->right = new TreeNode{6, new TreeNode{5,nullptr,nullptr}, new TreeNode{7,nullptr,nullptr}};

    cout << "Inorder: "; inorder(root); cout << endl;
    cout << "Preorder: "; preorder(root); cout << endl;
    cout << "Postorder: "; postorder(root); cout << endl;
    return 0;
}
