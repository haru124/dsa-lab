def generate_subsets(S):
    results = []

    def backtrack(index, path):
        if index == len(S):
            results.append(path[:])
            return
        # Include S[index]
        path.append(S[index])
        backtrack(index+1, path)
        path.pop()
        # Exclude S[index]
        backtrack(index+1, path)

    backtrack(0, [])
    return results

# Example
print("Subsets:", generate_subsets([1,2,3]))
# [[], [3], [2], [2,3], [1], [1,3], [1,2], [1,2,3]]
