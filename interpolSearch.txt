#include <iostream>
using namespace std;

int interpolationSearch(int arr[], int n, int key) {
    //for sorted array
    int low = 0;
    int high = n - 1;
    //cout << "\nKey: " <<key<< endl;
    //cout << "\nlow: " <<low<< endl;
    //cout << "\nhigh: " <<high<< endl;
    while (low <= high && key >= arr[low] && key <= arr[high]) {
        if (low == high) {
            if (arr[low] == key)
                return low;
            return -1;
        }

        // Interpolation formula to estimate the position
        int pos = low + ((key - arr[low]) * (high - low) / (arr[high] - arr[low]));
        //cout << "\npos: " <<pos;
        //cout << "\narr["<<pos<<"]: " <<arr[pos]<< endl;
        if (arr[pos] == key)
            return pos;

        else if (arr[pos] < key)
            low = pos + 1;
        else
            high = pos - 1;
    }

    return -1;
}

int main() {
    int n;
    //cout << "Enter the number of elements: ";
    cin >> n;

    int arr[n+1];
    //cout << "Enter the elements in sorted order: ";
    for (int i = 0; i <= n; i++) {
        cin >> arr[i];
    }

    int key, position;
    //cout << "Enter the key element to search: ";
    cin >> key;
    //cout << "\ninput Key: " <<key<< endl;
    

    position = interpolationSearch(arr, n+1, key);
    //cout << "\nPosition: " <<position<< endl;

    if (position == -1) {
        cout << "Key not found" << endl;
    } 
    else {
        cout << "Element " << key << " found in position " << position+1<< endl;
    }

    return 0;
}
