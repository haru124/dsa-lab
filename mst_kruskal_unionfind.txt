# Python 3
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0]*n

    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]

    def union(self, x, y):
        xr, yr = self.find(x), self.find(y)
        if xr == yr:
            return False
        if self.rank[xr] < self.rank[yr]:
            self.parent[xr] = yr
        elif self.rank[xr] > self.rank[yr]:
            self.parent[yr] = xr
        else:
            self.parent[yr] = xr
            self.rank[xr] += 1
        return True

def kruskal(n, edges):
    edges.sort(key=lambda x: x[2])  # sort by weight
    uf = UnionFind(n)
    mst = []
    cost = 0
    for u, v, w in edges:
        if uf.union(u, v):
            mst.append((u, v, w))
            cost += w
    return mst, cost

# Example usage:
n = 4
edges = [(0,1,1), (0,2,4), (1,2,2), (1,3,3), (2,3,5)]
mst, total_cost = kruskal(n, edges)
print("MST edges:", mst)
print("Total cost:", total_cost)
