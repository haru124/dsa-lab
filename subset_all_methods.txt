/////////////////////////////////////////////////////////
// Method 1: Recursive (Backtracking) - Stores subsets //
/////////////////////////////////////////////////////////
void recursiveHelper(const vector<int>& arr, int index, vector<int>& current, vector<vector<int>>& result) {
    // Base case: reached end of array
    if (index == arr.size()) {
        result.push_back(current);  // store the current subset
        return;
    }

    // Include current element
    current.push_back(arr[index]);
    recursiveHelper(arr, index + 1, current, result);

    // Exclude current element (backtrack)
    current.pop_back();
    recursiveHelper(arr, index + 1, current, result);
}

vector<vector<int>> generateSubsetsRecursive(const vector<int>& arr) {
    vector<vector<int>> result;
    vector<int> current;
    recursiveHelper(arr, 0, current, result);
    return result;
}

// Method 2: Iterative (Bitmasking) - Stores all subsets  //
////////////////////////////////////////////////////////////
vector<vector<int>> generateSubsetsBitmask(const vector<int>& arr) {
    int n = arr.size();
    int totalSubsets = 1 << n;  // 2^n subsets
    vector<vector<int>> result;

    for (int mask = 0; mask < totalSubsets; ++mask) {
        vector<int> subset;

        // Check each bit position
        for (int j = 0; j < n; ++j) {
            if (mask & (1 << j)) {
                subset.push_back(arr[j]);
            }
        }

        result.push_back(subset);  // store subset
    }

    return result;
}

/////////////////////////////////////////////////////////////////////////
// Method 3: Functional Style Recursion (Python-style) - Stores result //
/////////////////////////////////////////////////////////////////////////
vector<vector<int>> generateSubsetsFunctional(vector<int> arr) {
    if (arr.empty()) return {{}};

    int first = arr[0];
    arr.erase(arr.begin());  // reduce input

    vector<vector<int>> withoutFirst = generateSubsetsFunctional(arr);
    vector<vector<int>> withFirst;

    // Add 'first' to each subset generated from remaining
    for (auto subset : withoutFirst) {
        subset.insert(subset.begin(), first);
        withFirst.push_back(subset);
    }

    // Combine both
    withoutFirst.insert(withoutFirst.end(), withFirst.begin(), withFirst.end());
    return withoutFirst;
}

////////////////////////////////////////////////////////////
// Method 4: BFS Style using Queue - Stores all subsets   //
////////////////////////////////////////////////////////////
vector<vector<int>> generateSubsetsBFS(vector<int> arr) {
    vector<vector<int>> result = {{}};

    for (int num : arr) {
        int n = result.size();
        for (int i = 0; i < n; ++i) {
            vector<int> subset = result[i];
            subset.push_back(num);
            result.push_back(subset);
        }
    }

    return result;
}

/////////////////////////////////////////////////////////
// Utility: Print all subsets from vector<vector<int>> //
/////////////////////////////////////////////////////////
void printSubsets(const vector<vector<int>>& subsets, const string& title) {
    cout << "\n" << title << ":\n";
    for (const auto& subset : subsets) {
        cout << "{ ";
        for (int num : subset) {
            cout << num << " ";
        }
        cout << "}\n";
    }
}

////////////////////////////
// 🧪 Main driver function //
////////////////////////////
int main() {
    vector<int> arr = {1, 2, 3};

    // Method 1: Recursive Backtracking
    auto subsetsRecursive = generateSubsetsRecursive(arr);
    printSubsets(subsetsRecursive, "Recursive Backtracking");

    // Method 2: Iterative Bitmasking
    auto subsetsBitmask = generateSubsetsBitmask(arr);
    printSubsets(subsetsBitmask, "Iterative Bitmasking");

    // Method 3: Functional Recursion
    auto subsetsFunctional = generateSubsetsFunctional(arr);
    printSubsets(subsetsFunctional, "Functional Style Recursion");

    // Method 4: BFS using Queue
    auto subsetsBFS = generateSubsetsBFS(arr);
    printSubsets(subsetsBFS, "BFS using Queue");

    return 0;
}
